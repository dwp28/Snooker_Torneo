📋 Configurar Tabla en Supabase (Solo 1 vez)

Para que todos los espectadores y móviles vean las puntuaciones en vivo, sigue estos 3 pasos:

  1. Abre tu panel en supabase.com/dashboard
  2. En el menú de la izquierda, entra a SQL Editor -> New Query
  3. Pega el siguiente código y pulsa Run:
CREATE TABLE IF NOT EXISTS public.tournament_state (
    id TEXT PRIMARY KEY DEFAULT 'current',
    title TEXT NOT NULL DEFAULT 'Torneo Snooker: 1ª Copa de España',
    state JSONB NOT NULL,
    version INTEGER NOT NULL DEFAULT 1,
    last_updated TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

ALTER TABLE public.tournament_state ENABLE ROW LEVEL SECURITY;

DROP POLICY IF EXISTS "Permitir lectura publica a todos los usuarios" ON public.tournament_state;
DROP POLICY IF EXISTS "Permitir actualizacion del torneo" ON public.tournament_state;

CREATE POLICY "Permitir lectura publica a todos los usuarios"
ON public.tournament_state FOR SELECT USING (true);

CREATE POLICY "Permitir actualizacion del torneo"
ON public.tournament_state FOR ALL USING (true) WITH CHECK (true);

DO $$
BEGIN
  IF NOT EXISTS (
    SELECT 1 FROM pg_publication_tables 
    WHERE pubname = 'supabase_realtime' 
    AND schemaname = 'public' 
    AND tablename = 'tournament_state'
  ) THEN
    ALTER PUBLICATION supabase_realtime ADD TABLE public.tournament_state;
  END IF;
END $$;
